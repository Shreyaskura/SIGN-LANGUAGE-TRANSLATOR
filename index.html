Database Engineering Architecture & SQL Queries

-- 3NF Normalized PostgreSQL Schema for Sign Language Translator
CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE gestures (
    gesture_id SERIAL PRIMARY KEY,
    sign_name VARCHAR(50) UNIQUE NOT NULL,
    category VARCHAR(30) NOT NULL,
    description TEXT
);

CREATE TABLE translation_history (
    history_id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(user_id) ON DELETE CASCADE,
    gesture_id INT REFERENCES gestures(gesture_id) ON DELETE SET NULL,
    confidence_score NUMERIC(5,2) NOT NULL,
    translated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Performance Indexing for DBE
CREATE INDEX idx_translation_user ON translation_history(user_id);
CREATE INDEX idx_translation_time ON translation_history(translated_at DESC);

-- Advanced SQL Window Query: Top Most Frequently Recognized Signs per User
SELECT 
    u.username,
    g.sign_name,
    g.category,
    COUNT(t.history_id) AS total_translations,
    ROUND(AVG(t.confidence_score), 2) AS avg_confidence,
    RANK() OVER (PARTITION BY u.user_id ORDER BY COUNT(t.history_id) DESC) as user_rank
FROM translation_history t
JOIN users u ON t.user_id = u.user_id
JOIN gestures g ON t.gesture_id = g.gesture_id
GROUP BY u.user_id, u.username, g.gesture_id, g.sign_name, g.category
HAVING COUNT(t.history_id) > 1
ORDER BY total_translations DESC;
